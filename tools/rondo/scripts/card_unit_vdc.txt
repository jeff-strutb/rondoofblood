# first byte with the registers and RAM its header names. Power on, RUN on the card's title, let the kernel start.

wait 2
call ex_rcron
wait 2
call ex_rcroff
wait 2
call ex_bgoff
wait 2
call ex_spron
wait 2
call ex_sproff
wait 2
call ex_dspon
wait 2
call ex_dspoff
wait 2
call ex_setrcr A=40 X=00
wait 2
call ex_setred A=00 X=01
wait 2
call ex_setwrt A=00 X=02
wait 2
call ex_setdma M=20FA:00,20FB:30,20FC:00,20FD:40,20FE:00,20FF:20
wait 2
call ex_binbcd A=2A X=05
wait 2
call ex_binbcd A=63 X=00

call ex_rnd
wait 2

# WriteVramList with carry clear, X $00. The list is staged in scratch zero page $2004-$200E, the pointer in
# zTemp00-01: address $0000, word $0041, $FD with high byte $02, word $0242, $FE with address $0001, word $0243, $FF.
call WriteVramList X=00 P=00 M=2000:04,2001:20,2004:00,2005:00,2006:41,2007:FD M=2008:02,2009:42,200A:FE,200B:01,200C:00,200D:43,200E:FF
wait 2
# WriteVramList with carry set: every word the blank tile $0110. Address $0000, one word, $FF.
call WriteVramList X=00 P=01 M=2000:04,2001:20,2004:00,2005:00,2006:41,2007:FF
wait 2

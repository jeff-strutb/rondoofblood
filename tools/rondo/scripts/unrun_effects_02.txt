# Unit calls of EndingEffectFadeC (224F:371, ending.c's EndingEffectFade as this overlay version has it: effects_unrun.c), which no script
# runs. The overlay is the one scene_02 loads (scene sector 224F); this script is its prefix, the same frames to the end
# of its last wait, and then calls the fade in slot 0 with the overlay's bank $6C mapped at $8000 for each call. The lists are the
# disc's own: table at $9410 (list numbers $80 and up), and the pointer at $9402, which the overlay's code fills while it runs and
# which is poked here to the table itself so that list number 0 takes the positive path to the table's first list.
include _continue_to.txt
poke 2081 04
press I 6
wait 900
random 1041 3000 8
poke 20A3 09
poke 209A 02
wait 2400
# the pointer at $9402 (overlay bank $6C offset $1402) to the table at $9410
poke 6C:1402 10
poke 6C:1403 94
# timer running down: A the timer, N and Z from the decrement
call EndingEffectFadeC X=00 MPR4=6C M=3C00:02,3C30:03
wait 2
# list number $80: first call after the timer ran out (round 1), the table's list 0; then the second list ($81)
call EndingEffectFadeC X=00 MPR4=6C M=3C00:02,3C0C:80,3C24:00,3C30:00
wait 2
call EndingEffectFadeC X=00 MPR4=6C M=3C00:02,3C0C:81,3C24:00,3C30:00
wait 2
# list number 0 through the pointer at $9402 (the positive path)
call EndingEffectFadeC X=00 MPR4=6C M=3C00:02,3C0C:00,3C24:00,3C30:00
wait 2
# the eighth round: the slot is given back
call EndingEffectFadeC X=00 MPR4=6C M=3C00:02,3C0C:80,3C24:07,3C30:00
wait 2

# Unit calls of System Card CD entries no game path reaches, on the disc in the drive: a seek, the sense request (which drops the extra data) and
# the error retries. _AX at 20F8, _BX at 20FA, _CX at 20FC (sector, high byte first at CL), _DX at 20FE.
wait 120
press RUN 6
wait 700
call cd_request_sense X=00
wait 2
call cd_seek A=00 X=00 M=20FC:00,20FD:00,20FE:10
wait 30
call cd_retry_after_error A=06 X=00
wait 2
call cd_retry_after_error A=02 X=00
wait 30

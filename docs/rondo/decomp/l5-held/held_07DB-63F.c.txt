
/*
 * @routine 07DB:63F FloatDownThenSettle
 * @run 70:B63F
 * WHAT IT DOES: Animates the object facing the player (list 4), makes its sound every fourth
 * frame (first screen only), moves it, and while it is on the first screen and low enough (Y low
 * byte $10 or more) halves its speed on even frames; when it has all but stopped (or is under $10)
 * it enters state 5 with step 0.
 * HOW: StepFramesFacingPlayer, SoundEveryFourthFrameB, ApplyVelocity, HalveObjectVelocity.
 * WAITS: no.
 * Entry: X object, D clear. Exit: as the last test, load or store left them (A 5 and N, Z clear
 * on entering state 5).
 */
ScreenSoundEnd FloatDownThenSettle(uint8_t slot)
{
    ScreenSoundEnd result;
    FrameStepEnd frames;
    AxisMove move;
    ArithExit halved;
    uint8_t high, low, beat, a, b, both;
    Ended e, rest;

    jsr_push(0xB644);
    frames = StepFramesFacingPlayer(slot, 4);
    jsr_pop();
    e = ended_then(ended_with_y(ended_kept(), 4), frames.end);
    jsr_push(0xB647);
    result = SoundEveryFourthFrameB(slot, ended_y_or(e, 4));
    jsr_pop();
    result.before = ended_then(e, result.before);
    jsr_push(0xB64A);
    move = ApplyVelocity(slot);
    jsr_pop();
    rest = ended_of_axis_move(move);
    high = RD8(OBJ->y_hi[slot]);
    rest = ended_then(rest, ended_load(high));
    if (high != 0) {
        result.end = ended_then(result.end, rest);
        return result;
    }
    low = RD8(OBJ->y_lo[slot]);
    rest = ended_then(rest, ended_compare(low, 0x10));
    if (low >= 0x10) {
        beat = RD8(OBJ->f3618[slot]);
        rest = ended_then(rest, ended_load((uint8_t)(beat & 1)));
        if ((beat & 1) != 0) {
            result.end = ended_then(result.end, rest);
            return result;
        }
        jsr_push(0xB660);
        halved = HalveObjectVelocity(slot);
        jsr_pop();
        rest = ended_then(rest, ended_of_arith(halved));
        a = RD8(OBJ->vy_hi[slot]);
        both = (uint8_t)(a | RD8(OBJ->vy_lo[slot]));
        rest = ended_then(rest, ended_load(both));
        if (both != 0) {
            a = RD8(OBJ->vy_hi[slot]);
            b = RD8(OBJ->vy_lo[slot]);
            both = (uint8_t)(a & b);
            rest = ended_then(rest, ended_compare(both, 0xFF));
            if (both != 0xFF) {
                result.end = ended_then(result.end, rest);
                return result;
            }
        }
    }
    WR8(OBJ->state[slot], 5);
    WR8(OBJ->f2800[slot], 0);
    result.end = ended_then(ended_then(result.end, rest), ended_load(5));
    return result;
}

void abi_FloatDownThenSettle(void)
{
    abi_screen_sound_end(FloatDownThenSettle(cpu.x));
}

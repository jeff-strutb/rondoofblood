
; @routine 07DB:63F FloatDownThenSettle
; @run 70:B63F
; WHAT IT DOES: Animates the object facing the player (list 4), makes its sound every fourth
; frame (first screen only), moves it, and while it is on the first screen and low enough (Y low
; byte $10 or more) halves its speed on even frames; when it has all but stopped (or is under $10)
; it enters state 5 with step 0.
; HOW: StepFramesFacingPlayer, SoundEveryFourthFrameB, ApplyVelocity, HalveObjectVelocity.
; WAITS: no.
; Entry: X object, D clear. Exit: as the last test, load or store left them (A 5 and N, Z clear
; on entering state 5).
FloatDownThenSettle:
    ldy #$04
    jsr StepFramesFacingPlayer
    jsr SoundEveryFourthFrameB
    jsr VecApplyVelocity
    lda a:OBJ_Y_HI,x
    bne FloatDownThenSettle_at0
    lda a:OBJ_Y_LO,x
    cmp #$10
    bcc FloatDownThenSettle_at1
    lda a:OBJ_3618,x
    and #$01
    bne FloatDownThenSettle_at0
    jsr $B553
    lda a:OBJ_VY_HI,x
    ora a:OBJ_VY_LO,x
    beq FloatDownThenSettle_at1
    lda a:OBJ_VY_HI,x
    and a:OBJ_VY_LO,x
    cmp #$FF
    bne FloatDownThenSettle_at0
FloatDownThenSettle_at1:
    lda #$05
    sta a:OBJ_27A8,x
    stz a:OBJ_2800,x
FloatDownThenSettle_at0:
    rts

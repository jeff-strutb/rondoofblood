# Unit calls of the System Card's backup memory entries: format, free, files, write (new file, in place, growing), delete.
# _AX at 20F8, _BX at 20FA, _CX at 20FC, _DX at 20FE. The name is the kernel's own at 6C9A, the data the card's ROM at E000.
wait 120
press RUN 6
wait 700
call bm_format A=21 M=20F8:AD,20F9:F8
wait 2
call bm_free A=00
wait 2
call bm_files A=00 M=20F8:01,20F9:00,20FA:00,20FB:32
wait 2
call bm_write A=00 M=20F8:9A,20F9:6C,20FA:00,20FB:E0,20FC:20,20FD:00 M=20FE:00,20FF:00
wait 2
call bm_files A=00 M=20F8:01,20F9:00,20FA:00,20FB:32
wait 2
call bm_free A=00
wait 2
call bm_write A=00 M=20F8:9A,20F9:6C,20FA:00,20FB:E0,20FC:08,20FD:00 M=20FE:04,20FF:00
wait 2
call bm_write A=00 M=20F8:9A,20F9:6C,20FA:00,20FB:E0,20FC:40,20FD:00 M=20FE:20,20FF:00
wait 2
call bm_files A=00 M=20F8:01,20F9:00,20FA:00,20FB:32
wait 2
call bm_files A=00 M=20F8:02,20F9:00,20FA:00,20FB:32
wait 2
call bm_delete A=00 M=20F8:9A,20F9:6C
wait 2
call bm_delete A=00 M=20F8:9A,20F9:6C
wait 2
call bm_free A=00
wait 30

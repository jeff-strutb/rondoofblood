
; @routine 07DB:4BE TurnAroundAndHalveSpeed
; @run 70:B4BE
; WHAT IT DOES: Keeps the object's facing in f3720, makes its f3670 from the low nibble of
; PlayerDistanceSum, picks a new speed and turn from that (PickSpeedAndTurnFromTableB), then
; reverses both speeds and halves them, sets life_timer to $0C, stops the animation and advances
; the step.
; HOW: PlayerDistanceSum, the pick, NegateVelocityX, NegateVelocityY, HalveObjectVelocity.
; WAITS: no.
; Entry: X object, C in (kept to the sum), D clear. Exit: A $0C and N, Z from the new step; C
; from the halve, V from the second negate.
TurnAroundAndHalveSpeed:
    lda a:OBJ_2AC0,x
    sta a:OBJ_3720,x
    jsr VecPlayerDistanceSum
    and #$0F
    sta a:OBJ_3670,x
    jsr PickSpeedAndTurnFromTableB
    jsr VecNegateVelocityX
    jsr VecNegateVelocityY
    jsr $B553
    lda #$0C
    sta a:OBJ_34B8,x
    stz a:OBJ_3408,x
    stz a:OBJ_3460,x
    inc a:OBJ_2800,x
    rts
